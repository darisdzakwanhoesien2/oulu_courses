df_wrangled['Dt_Customer'] = pd.to_datetime(df_wrangled['Dt_Customer'], format='%d-%m-%Y')

### Cell 52: Code
from sklearn.linear_model import Ridge
from sklearn.model_selection import GridSearchCV

# Define the parameter grid for alpha
param_grid = {'alpha': [0.001, 0.01, 0.1, 1, 10, 100]}

# Create a Ridge regression model
ridge_model = Ridge()

# Create a GridSearchCV object to find the best alpha
grid_search = GridSearchCV(ridge_model, param_grid, cv=5, scoring='neg_mean_squared_error')

# Fit the GridSearchCV object to the training data
grid_search.fit(X_train, y_train)

# Print the best alpha value
print(f"Best alpha: {grid_search.best_params_['alpha']}")

# Get the best model
best_ridge_model = grid_search.best_estimator_

# Make predictions on the test data using the best model
y_pred_ridge = best_ridge_model.predict(X_test)

### Cell 61: Code
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Select relevant features for clustering
features_for_clustering = ['Income', 'TotalSpending', 'Age', 'Education_Graduation', 'Marital_Status_Married']

# Ensure 'TotalSpending' is available in the dataframe
X_cluster = df_wrangled[features_for_clustering]

# Standardize the selected features
scaler = StandardScaler()
X_cluster_scaled = scaler.fit_transform(X_cluster)

# Instantiate a KMeans object with a suitable number of clusters (e.g., 5)
kmeans = KMeans(n_clusters=5, random_state=42) # 2, 4, 5

# Fit the K-means model to the standardized data
kmeans.fit(X_cluster_scaled)

# Obtain cluster labels for each data point
cluster_labels = kmeans.labels_

# Add the cluster labels as a new column in df_wrangled
df_wrangled['cluster_label'] = cluster_labels

# Analyze the characteristics of each cluster
cluster_analysis = df_wrangled.groupby('cluster_label')[features_for_clustering].mean()
display(cluster_analysis)

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score
import matplotlib.pyplot as plt

def load_and_clean_data(filepath: str): # Including impuation of missing values
    df = pd.read_csv(filepath, sep='\t')
    print("Missing values before handling:\n", df.isnull().sum())
    df.fillna(df.median(numeric_only=True), inplace=True)
    print("Missing values after handling:\n", df.isnull().sum())
    return df

def filter_outliers(df, column, method='iqr', lower_quantile=0.01, upper_quantile=0.99):
    if method == 'iqr':
        Q1 = df[column].quantile(0.25)
        Q3 = df[column].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
    elif method == 'percentile':
        lower_bound = df[column].quantile(lower_quantile)
        upper_bound = df[column].quantile(upper_quantile)
    elif method == 'zscore':
        mean = df[column].mean()
        std = df[column].std()
        lower_bound = mean - 3 * std
        upper_bound = mean + 3 * std
    else:
        raise ValueError("Unsupported method. Choose from: 'iqr', 'percentile', or 'zscore'")
    
    return df[(df[column] >= lower_bound) & (df[column] <= upper_bound)]

def preprocess_data(df: pd.DataFrame, outlier_method='iqr'):
    df['Age'] = 2024 - df['Year_Birth']
    df['Age'] = df['Age'].clip(upper=81)

    bins = [30, 40, 50, 60, 70, 80, 90]
    labels = ['20-30', '30-40', '40-50', '50-60', '70-80', '80+']
    df['Age_group'] = pd.cut(df['Age'], bins=bins, labels=labels)

    # Apply selected outlier filtering method
    df = filter_outliers(df, column='Income', method=outlier_method)
    df = df[df['Age'] <= 81]

    return df

def feature_engineering(df: pd.DataFrame):
    df = df.copy()
    # Sum of the Data
    df['TotalSpending'] = df[['MntWines', 'MntFruits', 'MntMeatProducts', 
                              'MntFishProducts', 'MntSweetProducts', 'MntGoldProds']].sum(axis=1)
    
    df['NumTotalPurchases'] = df[['NumWebPurchases', 'NumCatalogPurchases', 
                                  'NumStorePurchases', 'NumDealsPurchases']].sum(axis=1)

    df['Promotion'] = df[['AcceptedCmp1', 'AcceptedCmp2', 'AcceptedCmp3',  
                          'AcceptedCmp4', 'AcceptedCmp5', 'Response']].sum(axis=1)

    df['Children'] = df[['Kidhome', 'Teenhome']].sum(axis=1)

    # Feature Engineering
    df['YearsAsCustomer'] = 2025 - df['Year_Birth']
    df['Dt_Customer'] = pd.to_datetime(df['Dt_Customer'], format='%d-%m-%Y')
    latest_date = df['Dt_Customer'].max()
    df['RecencyOfPurchase'] = (latest_date - df['Dt_Customer']).dt.days

    # Ratios    
    df['PurchaseFrequency'] = df['NumTotalPurchases'] / df['YearsAsCustomer']
    ## Ratios of spending on different products
    df['WineSpendingRatio'] = df['MntWines'] / df['TotalSpending']
    df['MeatSpendingRatio'] = df['MntMeatProducts'] / df['TotalSpending']
    df['FruitsSpendingRatio'] = df['MntFruits'] / df['TotalSpending']
    df['FishSpendingRatio'] = df['MntFishProducts'] / df['TotalSpending']
    df['SweetSpendingRatio'] = df['MntSweetProducts'] / df['TotalSpending']
    df['GoldSpendingRatio'] = df['MntGoldProds'] / df['TotalSpending']
    ## Ratios of purchases from different channels
    df['WebPurchasesRatio'] = df['NumWebPurchases'] / df['NumTotalPurchases']
    df['CatalogPurchasesRatio'] = df['NumCatalogPurchases'] / df['NumTotalPurchases']
    df['StorePurchasesRatio'] = df['NumStorePurchases'] / df['NumTotalPurchases']
    df['DealsPurchasesRatio'] = df['NumDealsPurchases'] / df['NumTotalPurchases']

    # One-hot encoding
    labels = ['Education', 'Marital_Status', 'Age_group']
    df = pd.get_dummies(df, columns=labels, prefix=labels)
    
    # Interaction features
    if 'Education_Graduation' in df.columns and 'Marital_Status_Married' in df.columns:
        df['TotalSpending_Education_Graduation'] = df['TotalSpending'] * df['Education_Graduation']
        df['NumTotalPurchases_Marital_Status_Married'] = df['NumTotalPurchases'] * df['Marital_Status_Married']
    df['TotalSpending_Income'] = df['TotalSpending'] * df['Income']
    
    return df

def run_regression(df: pd.DataFrame, target: str, drop_columns: list = None):
    df_model = df.copy()
    
    if drop_columns:
        df_model = df_model.drop(columns=drop_columns, errors='ignore')

    # Drop rows with NaNs (mostly from ratio features)
    df_model.dropna(inplace=True)

    X = df_model.drop(target, axis=1)
    y = df_model[target]
    
    # Drop non-numeric columns if any slipped through
    X = X.select_dtypes(include=[np.number])
    
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.25, random_state=42)
    
    model = LinearRegression()
    model.fit(X_train, y_train)
    y_pred = model.predict(X_test)
    
    # Evaluation metrics
    mse = mean_squared_error(y_test, y_pred)
    rmse = np.sqrt(mse)
    r2 = r2_score(y_test, y_pred)
    mae = mean_absolute_error(y_test, y_pred)
    
    print(f"\nModel Evaluation:")
    print(f"RMSE: {rmse:.2f}")
    print(f"R-squared: {r2:.2f}")
    print(f"MAE: {mae:.2f}")
    
    # Feature importance
    if hasattr(model, 'coef_'):
        importance = pd.DataFrame({'Feature': X.columns, 'Importance': model.coef_})
        print("\nTop Features:")
        display(importance.sort_values(by='Importance', ascending=False).head(15))
    
    return model

import pandas as pd

# ========== RQ2: SEGMENTING HIGH-VALUE CUSTOMERS ==========

def cluster_high_value_customers(df, features, cluster_nums=[2, 4, 5], n_components=2):
    X = df[features]
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    pca = PCA(n_components=n_components)
    PCA_array = pca.fit_transform(X_scaled)

    cluster_analysis_list = []

    for k in cluster_nums:
        kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
        df[f'cluster_label_{k}'] = kmeans.fit_predict(PCA_array)
        summary = df.groupby(f'cluster_label_{k}')[features].mean().reset_index()
        summary['Clusters'] = k
        cluster_analysis_list.append(summary)

    return df, pd.concat(cluster_analysis_list)

# ========== RQ3: PROMOTION RESPONSE CLASSIFICATION ==========

def classify_promotion_response(df, features, target='Response'):
    X = df[features].select_dtypes(include=[np.number])
    y = df[target]
    X = X.fillna(0)

    X_train, X_test, y_train, y_test = train_test_split(X, y, stratify=y, random_state=42)

    clf = RandomForestClassifier(n_estimators=100, random_state=42)
    clf.fit(X_train, y_train)
    y_pred = clf.predict(X_test)

    report = classification_report(y_test, y_pred, output_dict=True)
    importance = pd.DataFrame({
        'Feature': X.columns,
        'Importance': clf.feature_importances_
    }).sort_values(by='Importance', ascending=False)

    # SHAP
    explainer = shap.Explainer(clf, X_train)
    shap_values = explainer(X_test)

    return clf, report, importance, shap_values

import matplotlib.pyplot as plt
from sklearn.tree import plot_tree

def visualize_random_forest_tree(clf, feature_names, class_names=['No', 'Yes'], tree_index=0, max_depth=3):
    """
    Visualizes a single decision tree from a trained RandomForestClassifier.

    Parameters:
    - clf: Trained RandomForestClassifier
    - feature_names: List of feature names used in training
    - class_names: List of class names (default ['No', 'Yes'])
    - tree_index: Index of the tree to visualize (default 0)
    - max_depth: Maximum depth to display in the tree (default 3)
    """
    estimator = clf.estimators_[tree_index]

    plt.figure(figsize=(20, 10))
    plot_tree(
        estimator,
        feature_names=feature_names,
        class_names=class_names,
        filled=True,
        rounded=True,
        max_depth=max_depth,
        fontsize=10
    )
    plt.title(f"Visualization of Tree #{tree_index} (Max Depth: {max_depth})", fontsize=14)
    plt.show()
