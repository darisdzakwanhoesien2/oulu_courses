def bootstrap(D,M):

    # YOUR CODE HERE
    # Convert D to a numpy array
    D = np.array(D)
    
    # Number of observations in the original dataset
    N = len(D)

    # Generate random indices for bootstrapped samples
    indices = np.random.randint(0, N, size=(M, N))

    # Extract bootstrapped samples using the indices
    bootstrapped_samples = np.take(D, indices)

def pca_function(data):
    # Compute the mean vector mu
    mu = np.mean(data, axis=0).values
    
    #Compute the covariance matrix sigma by subtracting the mean from the center data     
    centered_data = data - mu
    N = len(data)
    sigma = (centered_data.T @ centered_data) / (N - 1)
    
    # Calculate the eigenvalues and eigenvectors of the covariance matrix
    lmbda, U = np.linalg.eig(sigma)

    # Pick the first principal component, corresponding to the largest eigenvalue
    # Sort the eigenvalues and eigenvectors in descending order of eigenvalues
    sorted_indices = np.argsort(lmbda)[::-1]
    lmbda = lmbda[sorted_indices]
    U = U[:, sorted_indices]
    
    # The first principal component
    vec1 = U[:, 0]
    
    # The second principal component (if needed)
    vec2 = U[:, 1]
    
    # Calculate endpoints of the line representing the first principal component
    vec1_end1 = mu + vec1
    vec1_end2 = mu - vec1

    # Plot the data points using a scatter plot
    plt.scatter(data[0], data[1], label='Data points', color='blue')

    # Plot the line representing the first principal component
    plt.plot([mu[0], vec1_end1[0]], [mu[1], vec1_end1[1]], color='red', label='First Principal Component')
    plt.plot([mu[0], vec1_end2[0]], [mu[1], vec1_end2[1]], color='red')

    # Plot mean
    plt.scatter(mu[0], mu[1], color='green', marker='o', label='Mean')

    # Add legend and labels
    plt.xlabel('X1')
    plt.ylabel('X2')
    plt.title('PCA - First Principal Component')
    plt.legend()
    plt.grid()
    plt.show()
    
    return mu, sigma, lmbda, U, vec1_end1, vec1_end2

def cov_matrix(X, mu):
    diff = X - mu
    return np.dot(diff.T, diff) / (len(X) - 1)

def pca_process(data):
    X = data.iloc[:, :2].values  # x and y coordinates
    y = data.iloc[:, 2].values   # class labels (0 or 1)

    # Compute the mean vectors for each class
    mu0 = np.mean(X[y == 0], axis=0)
    mu1 = np.mean(X[y == 1], axis=0)

    # Compute the global mean vector
    muG = np.mean(X, axis=0)

    # Compute the covariance matrices for each class without using np.cov()
    sigma0 = cov_matrix(X[y == 0], mu0)
    sigma1 = cov_matrix(X[y == 1], mu1)

    # Compute the within-class scatter matrix
    sigmaW = sigma0 + sigma1

    # Compute the between-class scatter matrix
    diff_mu0 = mu0 - muG
    diff_mu1 = mu1 - muG
    sigmaB = (len(X[y == 0]) * np.outer(diff_mu0, diff_mu0) + 
              len(X[y == 1]) * np.outer(diff_mu1, diff_mu1))

    # Compute the discriminant matrix H
    H = np.linalg.inv(sigmaW).dot(sigmaB)

    # Compute the eigenvalues and eigenvectors of the discriminant matrix H
    lmbda, U = np.linalg.eig(H)
    
    # Sort the eigenvectors by eigenvalue in descending order
    sorted_indices = np.argsort(lmbda)[::-1]
    lmbda = lmbda[sorted_indices]
    U = U[:, sorted_indices]
    
    # Extract the eigenvector (u) corresponding to the largest eigenvalue
    u = U[:, 0]

    # Calculate the line endpoints for plotting
    vec1 = muG + u
    vec2 = muG - u

    # Plot the data points and the LDA subspace
    plt.scatter(X[y == 0][:, 0], X[y == 0][:, 1], label='Class 0', c='blue', marker='o')
    plt.scatter(X[y == 1][:, 0], X[y == 1][:, 1], label='Class 1', c='red', marker='x')
    plt.plot([vec1[0], vec2[0]], [vec1[1], vec2[1]], c='green', label='LDA Subspace')
    plt.scatter(muG[0], muG[1], c='black', marker='s', label='Global Mean ($\mu_G$)')
    plt.title('LDA on Data')
    plt.xlabel('X coordinate')
    plt.ylabel('Y coordinate')
    plt.legend()   
    plt.grid(True)
    plt.show()

    return mu0, mu1, muG, sigma0, sigma1, sigmaW, sigmaB, H, lmbda, U, u, vec1, vec2

data_lda = pd.read_csv('data/lda_data.csv',header=None)
mu0, mu1, muG, sigma0, sigma1, sigmaW, sigmaB, H, lmbda, U, u, vec1, vec2 = pca_process(data_lda)
